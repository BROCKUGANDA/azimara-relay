    18→const REFERENCE_ALPHABET = '23456789ABCDEFGHJKMNPQRSTUVWXYZ'

    20→function newReference(prefix: string): string {
    21→  let suffix = ''
    22→  const bytes = crypto.getRandomValues(new Uint8Array(6))
    23→  for (const b of bytes) suffix += REFERENCE_ALPHABET[b % REFERENCE_ALPHABET.length]
    24→  return `${prefix}-${suffix}`
    25→}

    32→function parseRules(rulesJson: string): PolicyRule[] {
    33→  try {
    34→    const parsed = JSON.parse(rulesJson) as PolicyRule[]
    35→    return Array.isArray(parsed) && parsed.length > 0 ? parsed : DEFAULT_RULES
    36→  } catch {
    37→    return DEFAULT_RULES
    38→  }

    54→async function assetDecimals(mint: string): Promise<number> {
    55→  if (mint === SOL_MINT) return SOL_DECIMALS
    56→  const asset = await db.supportedAsset.findUnique({ where: { mint } })
    57→  if (!asset || !asset.enabled) {
    58→    throw new ApiError('UNSUPPORTED_ASSET', `Token mint ${mint} is not in the supported-asset registry`, 400)
    59→  }
    60→  return asset.decimals
    61→}

    63→async function transition(payment: PaymentRequest, to: string, opts?: { reason?: string; version?: number; data?: Record<string, unknown> }): Promise<PaymentRequest> {
    64→  assertTransition(payment.status, to)
    65→  const updated = await db.paymentRequest.updateMany({
    66→    where: { id: payment.id, version: opts?.version ?? payment.version },
    67→    data: { status: to, statusReason: opts?.reason ?? null, version: payment.version + 1, updatedAt: new Date() },
    68→  })
    69→  if (updated.count === 0) {
    70→    throw new ApiError('STALE_STATE', 'The payment was modified by another operation. Reload and retry.', 409)
    71→  }
    72→  await audit({
    73→    orgId: payment.orgId,
    74→    actorType: 'SYSTEM',
    75→    action: 'PAYMENT_STATE_TRANSITION',
    76→    resourceType: 'PaymentRequest',
    77→    resourceId: payment.id,
    78→    data: { from: payment.status, to, reason: opts?.reason, ...opts?.data },
    79→  })
    80→  return { ...payment, status: to, version: payment.version + 1 }
    81→}

    83→export async function createPayment(
    84→  orgId: string,
    85→  actorId: string,
    86→  input: CreatePaymentInput,
    87→): Promise<{ payment: PaymentRequest; idempotentReplay: boolean }> {
    88→  // Idempotency: same key → same payment, never a duplicate transfer.
    89→  // Records are stored under a per-org hashed key; lookups MUST use the same
    90→  // derivation or replays would never match.
    91→  const keyHash = input.idempotencyKey ? sha256(`${orgId}:${input.idempotencyKey}`) : null
    92→  if (keyHash) {
    93→    const existing = await db.idempotencyRecord.findUnique({ where: { key: keyHash } })
    94→    if (existing && existing.orgId === orgId) {
    95→      if (existing.status === 'COMPLETED' && existing.responseJson) {
    96→        const snapshot = JSON.parse(existing.responseJson) as { paymentId: string }
    97→        const payment = await db.paymentRequest.findUnique({ where: { id: snapshot.paymentId } })
    98→        if (payment) return { payment, idempotentReplay: true }
    99→      }
   100→      if (existing.status === 'IN_FLIGHT' && Date.now() - new Date(existing.updatedAt).getTime() < 120_000) {
   101→        throw new ApiError('OPERATION_IN_FLIGHT', 'An identical payment creation is already in progress.', 409)
   102→      }
   103→    }
   104→  }
   105→
   106→  const ent = entitlementsFor((await db.organization.findUniqueOrThrow({ where: { id: orgId } })).plan)
   107→  const monthStart = new Date(new Date().getFullYear(), new Date().getMonth(), 1)
   108→  const monthlyCount = await db.paymentRequest.count({ where: { orgId, createdAt: { gte: monthStart } } })
   109→  const { assertWithinLimit } = await import('@/lib/entitlements')
   110→  assertWithinLimit(ent, 'monthlyPayments', monthlyCount)
   111→
   112→  const wallet = await db.wallet.findUnique({ where: { id: input.sourceWalletId } })
   113→  if (!wallet || wallet.orgId !== orgId || wallet.revokedAt) {
   114→    throw new ApiError('INVALID_WALLET', 'Source wallet not found or not available for this organization', 400)
   115→  }
   116→  if (wallet.ownershipStatus !== 'VERIFIED') {
   117→    throw new ApiError('WALLET_NOT_VERIFIED', 'Source wallet ownership must be verified before payments', 400)
   118→  }
   119→  const decimals = await assetDecimals(input.mint)
   120→  let amountBase: string
   121→  try {
   122→    amountBase = (await import('@/lib/money')).parseAmountToBaseUnits(input.amount, decimals)
   123→  } catch (err) {
   124→    throw new ApiError('INVALID_AMOUNT', err instanceof Error ? err.message : 'Invalid amount', 400)
   125→  }
   126→  assertBaseUnits(amountBase)
   127→  if (assertBaseUnits(amountBase) === 0n) throw new ApiError('INVALID_AMOUNT', 'Payment amount must be greater than zero', 400)
   128→  if (!isValidSolanaAddress(input.destination)) {
   129→    throw new ApiError('INVALID_DESTINATION', 'Destination is not a valid Solana address', 400)
   130→  }
   131→
   132→  let recipient: Prisma.RecipientGetPayload<{ include: { changes: true } }> | null = null
   133→  if (input.recipientId) {
   134→    recipient = await db.recipient.findFirst({ where: { id: input.recipientId, orgId }, include: { changes: true } })
   135→    if (!recipient) throw new ApiError('RECIPIENT_NOT_FOUND', 'Recipient not found in this organization', 404)
   136→    if (recipient.status !== 'ACTIVE') throw new ApiError('RECIPIENT_INACTIVE', `Recipient is ${recipient.status.toLowerCase()}`, 400)
   137→    if (recipient.address !== input.destination) {
   138→      throw new ApiError('RECIPIENT_ADDRESS_MISMATCH', 'Destination does not match the registered recipient address. Recipient addresses cannot be silently changed.', 400)
   139→    }
   140→  }
   141→
   142→  // Duplicate-request detection: same wallet, destination, mint and amount
   143→  // already pending or recently completed.
   144→  const duplicate = await db.paymentRequest.findFirst({
   145→    where: {
   146→      orgId,
   147→      sourceWalletId: wallet.id,
   148→      destination: input.destination,
   149→      mint: input.mint,
   150→      amountBaseUnits: amountBase,
   151→      status: { in: ['PENDING_APPROVAL', 'APPROVED', 'QUOTED', 'AWAITING_SIGNATURE', 'PROCESSING', 'FUNDING_BROADCAST', 'FUNDING_CONFIRMED', 'INITIALIZATION_BROADCAST'] },
   152→    },
   153→  })
   154→  if (duplicate) {
   155→    throw new ApiError('DUPLICATE_PAYMENT', `A payment with identical wallet, destination, token and amount already exists (${duplicate.reference}) and is ${duplicate.status}.`, 409, { duplicateReference: duplicate.reference })
   156→  }
   157→
   158→  const idemKey = input.idempotencyKey ?? `pay:${orgId}:${shortId()}`
   159→  const markerKeyHash = sha256(`${orgId}:${idemKey}`)
   160→
   161→  // In-flight marker with a deterministic key so concurrent double-clicks converge.
   162→  const marker = await db.idempotencyRecord.upsert({
   163→    where: { key: markerKeyHash },
   164→    create: { key: markerKeyHash, orgId, scope: 'PAYMENT_CREATE', requestHash: markerKeyHash, status: 'IN_FLIGHT' },
   165→    update: { status: 'IN_FLIGHT', updatedAt: new Date() },
   166→  })
   167→
   168→  try {
   169→    let payment = await db.paymentRequest.create({
   170→      data: {
   171→        orgId,
   172→        reference: newReference('PMT'),
   173→        type: input.source === 'API' ? 'API' : 'ONE_OFF',
   174→        status: 'DRAFT',
   175→        sourceWalletId: wallet.id,
   176→        recipientId: recipient?.id ?? null,
   177→        destination: input.destination,
   178→        mint: input.mint,
   179→        amountBaseUnits: amountBase,
   180→        reason: input.reason ?? null,
   181→        invoiceRef: input.invoiceRef ?? null,
   182→        requestedExecutionAt: input.requestedExecutionAt ? new Date(input.requestedExecutionAt) : null,
   183→        source: input.source ?? 'APP',
   184→        idempotencyKey: idemKey,
   185→        createdBy: actorId,
   186→      },
   187→    })
   188→    payment = await transition(payment, 'VALIDATING')
   189→
   190→    // Policy evaluation determines how many independent approvals are needed.
   191→    const policy = await getActivePolicy(orgId)
   192→    const rules = parseRules(policy?.rulesJson ?? '[]')
   193→    const evaluation = evaluateRules(rules, payment, recipient)
   194→    const multiOk = ent.multiApprover || evaluation.approversRequired <= 1
   195→    payment = await db.paymentRequest.update({
   196→      where: { id: payment.id },
   197→      data: { approvalsRequired: multiOk ? evaluation.approversRequired : 1 },
   198→    })
   199→    await db.approvalEvent.create({
   200→      data: {
   201→        paymentId: payment.id,
   202→        type: 'POLICY_EVALUATED',
   203→        actorId,
   204→        dataJson: JSON.stringify({ ...evaluation, policyId: policy?.id ?? null, policyVersion: policy?.version ?? null, cappedByPlan: !multiOk }),
   205→      },
   206→    })
   207→    payment = await transition(payment, 'PENDING_APPROVAL', { reason: evaluation.matchedRules.map((r) => r.id).join('+') })
   208→    await notify(orgId, {
   209→      type: 'APPROVAL_REQUESTED',
   210→      title: `Payment ${payment.reference} awaiting approval`,
   211→      body: `${evaluation.approversRequired} approver(s) required — ${evaluation.matchedRules.map((r) => r.description).join('; ')}`,
   212→      link: `#/app/payments/${payment.id}`,
   213→    })
   214→    await db.idempotencyRecord.update({ where: { key: markerKeyHash }, data: { status: 'COMPLETED', responseJson: JSON.stringify({ paymentId: payment.id }) } })
   215→    return { payment, idempotentReplay: false }
   216→  } catch (err) {
   217→    await db.idempotencyRecord.update({ where: { key: markerKeyHash }, data: { status: 'FAILED', responseJson: JSON.stringify({ error: String(err) }) } }).catch(() => undefined)
   218→    throw err
   219→  }
   220→}

   222→export async function approvePayment(paymentId: string, orgId: string, approverId: string, approverRole: string, decision: 'APPROVED' | 'REJECTED', reason?: string) {
   223→  const payment = await db.paymentRequest.findFirst({ where: { id: paymentId, orgId } })
   224→  if (!payment) throw new ApiError('NOT_FOUND', 'Payment not found', 404)
   225→  if (payment.status !== 'PENDING_APPROVAL') {
   226→    throw new ApiError('INVALID_STATE', `Payment is ${payment.status}, not awaiting approval`, 409)
   227→  }
   228→  if (payment.createdBy === approverId && payment.approvalsRequired > 1) {
   229→    throw new ApiError('SEPARATION_OF_DUTIES', 'The creator of a multi-approver payment cannot be one of its approvers', 403)
   230→  }
   231→  const hash = materialPayload(payment)
   232→  const existing = await db.paymentApproval.findMany({ where: { paymentId, decision: { in: ['APPROVED', 'REJECTED'] } } })
   233→  if (existing.some((a) => a.approverId === approverId)) {
   234→    throw new ApiError('ALREADY_DECIDED', 'You have already recorded a decision on this payment', 409)
   235→  }
   236→  if (decision === 'APPROVED') {
   237→    // A prior approval whose payload hash no longer matches is invalid.
   238→    const valid = existing.filter((a) => a.decision === 'APPROVED' && a.payloadHash === hash && (!a.expiresAt || a.expiresAt > new Date()))
   239→    await db.paymentApproval.create({
   240→      data: {
   241→        paymentId,
   242→        orgId,
   243→        approverId,
   244→        policyId: null,
   245→        decision,
   246→        reason: reason ?? null,
   247→        payloadHash: hash,
   248→        evidenceJson: JSON.stringify({ payloadFields: { destination: payment.destination, mint: payment.mint, amountBaseUnits: payment.amountBaseUnits }, decidedAt: new Date().toISOString() }),
   249→        expiresAt: new Date(Date.now() + 7 * 24 * 3600 * 1000),
   250→      },
   251→    })
   252→    await db.approvalEvent.create({ data: { paymentId, type: 'APPROVED', actorId: approverId, dataJson: JSON.stringify({ payloadHash: hash }) } })
   253→    const approvalsNow = await db.paymentApproval.count({ where: { paymentId, decision: 'APPROVED', payloadHash: hash, expiresAt: { gt: new Date() } } })
   254→    const distinct = new Set((await db.paymentApproval.findMany({ where: { paymentId, decision: 'APPROVED', payloadHash: hash } })).map((a) => a.approverId)).size
   255→    if (Math.min(distinct, approvalsNow) >= payment.approvalsRequired) {
   256→      const fresh = await db.paymentRequest.findUniqueOrThrow({ where: { id: paymentId } })
   257→      await transition(fresh, 'APPROVED')
   258→    }
   259→  } else {
   260→    await db.paymentApproval.create({
   261→      data: { paymentId, orgId, approverId, decision, reason: reason ?? null, payloadHash: hash, evidenceJson: '{}' },
   262→    })
   263→    await db.approvalEvent.create({ data: { paymentId, type: 'REJECTED', actorId: approverId, dataJson: JSON.stringify({ reason }) } })
   264→    await transition(payment, 'REJECTED', { reason })
   265→  }
   266→  await audit({ orgId, actorId: approverId, action: `PAYMENT_${decision}`, resourceType: 'PaymentRequest', resourceId: paymentId })
   267→  return db.paymentRequest.findUniqueOrThrow({ where: { id: paymentId } })
   268→}

   270→export async function paymentApprovalState(payment: PaymentRequest) {
   271→  const approvals = await db.paymentApproval.findMany({ where: { paymentId: payment.id }, include: { approver: { select: { name: true, email: true } } } })
   272→  const hash = materialPayload(payment)
   273→  const validApprovals = approvals.filter((a) => a.decision === 'APPROVED' && a.payloadHash === hash && (!a.expiresAt || a.expiresAt > new Date()))
   274→  return {
   275→    approvals,
   276→    validCount: new Set(validApprovals.map((a) => a.approverId)).size,
   277→    required: payment.approvalsRequired,
   278→    payloadHash: hash,
   279→    isSatisfied: new Set(validApprovals.map((a) => a.approverId)).size >= payment.approvalsRequired && payment.status === 'APPROVED',
   280→  }
   281→}

   283→export async function requestQuote(paymentId: string, orgId: string): Promise<{ ok: true; quoteId: string } | { ok: false; setupRequired: boolean; detail: string }> {
   284→  let payment = await db.paymentRequest.findFirst({ where: { id: paymentId, orgId } })
   285→  if (!payment) throw new ApiError('NOT_FOUND', 'Payment not found', 404)
   286→  if (payment.status !== 'APPROVED' && payment.status !== 'QUOTED' && payment.status !== 'QUOTE_EXPIRED') {
   287→    throw new ApiError('INVALID_STATE', `Quoting requires an approved payment (current: ${payment.status})`, 409)
   288→  }
   289→  payment = await transition(payment, 'QUOTING')
   290→  const integration = await db.providerIntegration.findUnique({ where: { orgId_provider: { orgId, provider: 'MULTIHOPPER' } } })
   291→  const client = new MultiHopperClient(providerConfigFromRow(integration))
   292→  try {
   293→    const { raw, quoteId } = await client.estimate({
   294→      sourceMint: payment.mint,
   295→      amountBaseUnits: payment.amountBaseUnits,
   296→      destinations: [{ address: payment.destination }],
   297→    })
   298→    const responseJson = JSON.stringify(raw)
   299→    // Preserve the provider quote verbatim; parsed fee fields are stored
   300→    // separately and only derived from documented response fields.
   301→    const principalBase = payment.amountBaseUnits
   302→    const quote = await db.paymentQuote.create({
   303→      data: {
   304→        paymentId: payment.id,
   305→        requestJson: JSON.stringify({ sourceMint: payment.mint, amountBaseUnits: payment.amountBaseUnits, destinations: [{ address: payment.destination }] }),
   306→        responseJson,
   307→        principalBase,
   308→        feePctBps: null,
   309→        recipientGetsBase: principalBase,
   310→        expiresAt: new Date(Date.now() + 5 * 60 * 1000),
   311→        status: 'ACTIVE',
   312→      },
   313→    })
   314→    await db.paymentQuote.updateMany({ where: { paymentId: payment.id, id: { not: quote.id }, status: 'ACTIVE' }, data: { status: 'SUPERSEDED' } })
   315→    const fresh = await db.paymentRequest.findUniqueOrThrow({ where: { id: payment.id } })
   316→    await transition(fresh, 'QUOTED', { reason: quoteId })
   317→    await audit({ orgId, action: 'PAYMENT_QUOTED', resourceType: 'PaymentRequest', resourceId: payment.id, actorType: 'SYSTEM' })
   318→    return { ok: true, quoteId: quote.id }
   319→  } catch (err) {
   320→    if (err instanceof SetupRequiredError) {
   321→      const fresh = await db.paymentRequest.findUniqueOrThrow({ where: { id: payment.id } })
   322→      await db.paymentRequest.update({ where: { id: fresh.id }, data: { statusReason: 'PROVIDER_SETUP_REQUIRED' } })
   323→      await transition(fresh, 'APPROVED', { reason: 'PROVIDER_SETUP_REQUIRED — quote not obtained; payment held at APPROVED' })
   324→      await notify(orgId, {
   325→        type: 'INTEGRATION_SETUP_REQUIRED',
   326→        title: 'MultiHopper credentials required',
   327→        body: `Payment ${payment.reference} is approved but cannot be quoted until MultiHopper ${integration?.environment ?? 'DEVNET'} credentials are configured.`,
   328→        link: '#/app/integrations',
   329→      })
   330→      return { ok: false, setupRequired: true, detail: err.detail }
   331→    }
   332→    if (err instanceof MultiHopperError) {
   333→      const fresh = await db.paymentRequest.findUniqueOrThrow({ where: { id: payment.id } })
   334→      await db.paymentRequest.update({ where: { id: fresh.id }, data: { lastError: `${err.code}: ${err.message}` } })
   335→      await transition(fresh, 'MANUAL_REVIEW', { reason: `QUOTE_FAILED: ${err.code}` })
   336→      return { ok: false, setupRequired: false, detail: err.message }
   337→    }
   338→    throw err
   339→  }
   340→}

   342→/// Creates the provider transfer (idempotent by stable externalId) and moves
   343→/// the payment to AWAITING_SIGNATURE. Reuses the same externalId on retry so
   344→/// an unknown creation outcome can never yield two transfers.
   345→export async function startExecution(paymentId: string, orgId: string, actorId: string) {
   346→  let payment = await db.paymentRequest.findFirst({ where: { id: paymentId, orgId }, include: { quotes: { where: { status: 'ACTIVE' } } } })
   347→  if (!payment) throw new ApiError('NOT_FOUND', 'Payment not found', 404)
   348→  if (payment.status !== 'QUOTED') throw new ApiError('INVALID_STATE', `Execution requires a fresh quote (current: ${payment.status})`, 409)
   349→  const activeQuote = payment.quotes[0]
   350→  if (!activeQuote || activeQuote.expiresAt < new Date()) {
   351→    await transition(payment, 'QUOTE_EXPIRED', { reason: 'Quote expired before signing' })
   352→    throw new ApiError('QUOTE_EXPIRED', 'The provider quote expired. Request a new quote to continue.', 409)
   353→  }
   354→  const approvalState = await paymentApprovalState(payment)
   355→  if (!approvalState.isSatisfied && payment.status === 'QUOTED') {
   356→    throw new ApiError('APPROVAL_INVALID', 'The approval payload no longer matches this payment. Re-approval required.', 409)
   357→  }
   358→  const integration = await db.providerIntegration.findUnique({ where: { orgId_provider: { orgId, provider: 'MULTIHOPPER' } } })
   359→  const client = new MultiHopperClient(providerConfigFromRow(integration))
   360→  const externalId = `relay-${payment.id}`
   361→  let transfer = await db.providerTransfer.findUnique({ where: { paymentId: payment.id } })
   362→  if (!transfer) {
   363→    transfer = await db.providerTransfer.create({ data: { orgId, paymentId: payment.id, externalId, provider: 'MULTIHOPPER', internalPhase: 'CREATING' } })
   364→  }
   365→  try {
   366→    if (!transfer.providerTransferId) {
   367→      const result = await client.createTransfer({
   368→        externalId,
   369→        sourceOwner: (await db.wallet.findUniqueOrThrow({ where: { id: payment.sourceWalletId! } })).address,
   370→        sourceMint: payment.mint,
   371→        amountBaseUnits: payment.amountBaseUnits,
   372→        destinations: [{ address: payment.destination }],
   373→        quoteId: undefined,
   374→        metadata: { orgReference: payment.reference },
   375→      })
   376→      transfer = await db.providerTransfer.update({
   377→        where: { id: transfer.id },
   378→        data: {
   379→          providerTransferId: result.providerTransferId,
   380→          providerStatus: 'CREATED',
   381→          lifecycleJson: JSON.stringify({ created: result.raw }),
   382→          internalPhase: 'CREATED',
   383→        },
   384→      })
   385→    }
   386→    const fresh = await db.paymentRequest.findUniqueOrThrow({ where: { id: payment.id } })
   387→    await transition(fresh, 'AWAITING_SIGNATURE')
   388→    await audit({ orgId, actorId, action: 'TRANSFER_CREATED', resourceType: 'ProviderTransfer', resourceId: transfer.id, data: { externalId, providerTransferId: transfer.providerTransferId } })
   389→    return { transfer, client }
   390→  } catch (err) {
   391→    if (err instanceof SetupRequiredError) {
   392→      await db.providerTransfer.update({ where: { id: transfer.id }, data: { internalPhase: 'BLOCKED_SETUP' } })
   393→      throw err
   394→    }
   395→    throw err
   396→  }
   397→}

   399→export async function prepareExecution(paymentId: string, orgId: string) {
   400→  const payment = await db.paymentRequest.findFirst({ where: { id: paymentId, orgId }, include: { transfer: true } })
   401→  if (!payment) throw new ApiError('NOT_FOUND', 'Payment not found', 404)
   402→  if (payment.status !== 'AWAITING_SIGNATURE' && payment.status !== 'FUNDING_BROADCAST') {
   403→    throw new ApiError('INVALID_STATE', `Prepare requires AWAITING_SIGNATURE (current: ${payment.status})`, 409)
   404→  }
   405→  if (!payment.transfer?.providerTransferId) throw new ApiError('TRANSFER_MISSING', 'No provider transfer reference for this payment', 409)
   406→  const integration = await db.providerIntegration.findUnique({ where: { orgId_provider: { orgId, provider: 'MULTIHOPPER' } } })
   407→  const client = new MultiHopperClient(providerConfigFromRow(integration))
   408→  const prepared = await client.prepareTransfer(payment.transfer.providerTransferId)
   409→  await db.providerTransfer.update({
   410→    where: { id: payment.transfer.id },
   411→    data: {
   412→      progressJson: JSON.stringify(prepared.bundles),
   413→      internalPhase: 'PREPARED',
   414→    },
   415→  })
   416→  return prepared.bundles
   417→}

   419→/// Intermediate confirmation: keeper funding signature + empty route-init list.
   420→export async function confirmKeeperFunding(paymentId: string, orgId: string, actorId: string, keeperSignature: string) {
   421→  const payment = await db.paymentRequest.findFirst({ where: { id: paymentId, orgId }, include: { transfer: true } })
   422→  if (!payment) throw new ApiError('NOT_FOUND', 'Payment not found', 404)
   423→  if (payment.status !== 'AWAITING_SIGNATURE') throw new ApiError('INVALID_STATE', `Current state ${payment.status} — expected AWAITING_SIGNATURE`, 409)
   424→  if (!payment.transfer?.providerTransferId) throw new ApiError('TRANSFER_MISSING', 'No provider transfer reference', 409)
   425→  await db.transactionSignature.create({ data: { paymentId, purpose: 'KEEPER_FUNDING', signature: keeperSignature, network: (await db.wallet.findUniqueOrThrow({ where: { id: payment.sourceWalletId! } })).network } })
   426→  const integration = await db.providerIntegration.findUnique({ where: { orgId_provider: { orgId, provider: 'MULTIHOPPER' } } })
   427→  const client = new MultiHopperClient(providerConfigFromRow(integration))
   428→  const result = await client.confirmBroadcast({ providerTransferId: payment.transfer.providerTransferId, keeperFundingSignature: keeperSignature, routeInitSignatures: [] })
   429→  await db.providerTransfer.update({
   430→    where: { id: payment.transfer.id },
   431→    data: { lifecycleJson: JSON.stringify({ keeperConfirmed: result.raw }), internalPhase: 'KEEPER_CONFIRMED', providerStatus: 'FUNDING_CONFIRMED' },
   432→  })
   433→  await transition(payment, 'FUNDING_BROADCAST', { reason: 'Keeper funding signed and submitted' })
   434→  await audit({ orgId, actorId, action: 'KEEPER_FUNDING_CONFIRMED', resourceType: 'PaymentRequest', resourceId: paymentId })
   435→  return result.raw
   436→}

   438→export async function confirmBroadcastFinal(paymentId: string, orgId: string, routeInitSignatures: string[]) {
   439→  const payment = await db.paymentRequest.findFirst({ where: { id: paymentId, orgId }, include: { transfer: true } })
   440→  if (!payment) throw new ApiError('NOT_FOUND', 'Payment not found', 404)
   441→  if (!['FUNDING_BROADCAST', 'FUNDING_CONFIRMED', 'INITIALIZATION_BROADCAST'].includes(payment.status)) {
   442→    throw new ApiError('INVALID_STATE', `Current state ${payment.status} — final confirmation is premature`, 409)
   443→  }
   444→  if (!payment.transfer?.providerTransferId) throw new ApiError('TRANSFER_MISSING', 'No provider transfer reference', 409)
   445→  const integration = await db.providerIntegration.findUnique({ where: { orgId_provider: { orgId, provider: 'MULTIHOPPER' } } })
   446→  const client = new MultiHopperClient(providerConfigFromRow(integration))
   447→  const result = await client.confirmBroadcast({ providerTransferId: payment.transfer.providerTransferId, routeInitSignatures })
   448→  for (const sig of routeInitSignatures) {
   449→    await db.transactionSignature.create({ data: { paymentId, purpose: 'ROUTE_INIT', signature: sig, network: 'devnet' } }).catch(() => undefined)
   450→  }
   451→  await db.providerTransfer.update({
   452→    where: { id: payment.transfer.id },
   453→    data: { lifecycleJson: JSON.stringify({ finalConfirmed: result.raw }), internalPhase: 'BROADCAST_CONFIRMED', providerStatus: 'PROCESSING' },
   454→  })
   455→  const fresh = await db.paymentRequest.findUniqueOrThrow({ where: { id: paymentId } })
   456→  await transition(fresh, 'PROCESSING', { reason: 'Final broadcast confirmation submitted' })
   457→  await db.jobRecord.create({ data: { type: 'TRANSFER_MONITOR', orgId, payloadJson: JSON.stringify({ paymentId }), dedupeKey: `monitor:${paymentId}`, runAt: new Date(Date.now() + 15_000) } }).catch(() => undefined)
   458→  return result.raw
   459→}

   461→/// Provider + chain reconciliation for one payment. Provider status maps into
   462→/// internal state; settlement verification requires both provider evidence and
   463→/// chain confirmation before COMPLETED.
   464→export async function syncTransfer(paymentId: string, orgId: string) {
   465→  const payment = await db.paymentRequest.findFirst({ where: { id: paymentId, orgId }, include: { transfer: true, wallet: true } })
   466→  if (!payment) return null
   467→  if (!payment.transfer?.providerTransferId) return null
   468→  const integration = await db.providerIntegration.findUnique({ where: { orgId_provider: { orgId, provider: 'MULTIHOPPER' } } })
   469→  const client = new MultiHopperClient(providerConfigFromRow(integration))
   470→  try {
   471→    const { raw, status } = await client.getTransfer(payment.transfer.providerTransferId)
   472→    await db.providerTransfer.update({
   473→      where: { id: payment.transfer.id },
   474→      data: { providerStatus: status ?? payment.transfer.providerStatus, lifecycleJson: JSON.stringify({ lastSync: raw, syncedAt: new Date().toISOString() }) },
   475→    })
   476→    await db.providerTransferEvent.create({ data: { transferId: payment.transfer.id, eventType: status ?? 'UNKNOWN', payloadJson: JSON.stringify(raw), source: 'POLL' } })
   477→    const mapped = mapProviderStatus(status)
   478→    const fresh = await db.paymentRequest.findUniqueOrThrow({ where: { id: paymentId } })
   479→    if (mapped === 'SETTLEMENT_VERIFICATION') {
   480→      // Require chain evidence before completion.
   481→      const sigs = await db.transactionSignature.findMany({ where: { paymentId } })
   482→      let chainConfirmed = false
   483→      let chainStatus: string | null = null
   484→      for (const sig of sigs) {
   485→        const st = await (await import('@/lib/solana')).getSignatureStatus(payment.wallet?.network ?? 'devnet', sig.signature)
   486→        if (st?.confirmed) {
   487→          chainConfirmed = true
   488→          chainStatus = st.status
   489→          await db.transactionSignature.update({ where: { id: sig.id }, data: { confirmedAt: new Date(), slot: st.slot } })
   490→          break
   491→        }
   492→      }
   493→      await transition(fresh, 'SETTLEMENT_VERIFICATION', { reason: 'Provider reports settlement; verifying chain evidence' })
   494→      if (chainConfirmed) {
   495→        await completeSettlement(fresh, status ?? 'COMPLETED', chainStatus ?? 'CONFIRMED')
   496→      } else {
   497→        await db.reconciliationRecord.upsert({
   498→          where: { paymentId },
   499→          create: { orgId, paymentId, providerRef: payment.transfer.providerTransferId, providerStatus: status, state: 'UNRECONCILED', detail: 'Provider settled; awaiting onchain confirmation evidence', expectedBase: payment.amountBaseUnits },
   500→          update: { providerStatus: status, state: 'UNRECONCILED', detail: 'Provider settled; awaiting onchain confirmation evidence', lastCheckedAt: new Date() },
   501→        })
   502→      }
   503→    } else if (typeof mapped === 'string' && ['FAILED', 'TRANSFER_EXPIRED', 'RECOVERABLE', 'REFUNDED', 'RESCUED'].includes(mapped)) {
   504→      await transition(fresh, mapped, { reason: `Provider status: ${status}` })
   505→      await db.reconciliationRecord.upsert({
   506→        where: { paymentId },
   507→        create: { orgId, paymentId, providerRef: payment.transfer.providerTransferId, providerStatus: status, state: mapped === 'FAILED' ? 'FAILED' : 'UNRECONCILED', detail: `Provider status ${status}`, expectedBase: payment.amountBaseUnits },
   508→        update: { providerStatus: status, state: mapped === 'FAILED' ? 'FAILED' : 'UNRECONCILED', detail: `Provider status ${status}`, lastCheckedAt: new Date() },
   509→      })
   510→      await notify(orgId, { type: 'TRANSFER_STATE', title: `Payment ${payment.reference}: ${mapped}`, body: `Provider status: ${status}`, link: `#/app/payments/${paymentId}` })
   511→    } else if (typeof mapped === 'string' && !['COMPLETED'].includes(mapped) && fresh.status !== mapped) {
   512→      await transition(fresh, mapped, { reason: `Provider status: ${status}` }).catch(() => undefined)
   513→    }
   514→    return { providerStatus: status, mapped }
   515→  } catch (err) {
   516→    if (err instanceof SetupRequiredError) return { providerStatus: null, mapped: null, setupRequired: true, detail: err.detail }
   517→    if (err instanceof MultiHopperError) {
   518→      await db.reconciliationRecord.upsert({
   519→        where: { paymentId },
   520→        create: { orgId, paymentId, providerRef: payment.transfer.providerTransferId, state: 'UNRECONCILED', detail: `Sync failed: ${err.code}`, expectedBase: payment.amountBaseUnits },
   521→        update: { detail: `Sync failed: ${err.code}`, lastCheckedAt: new Date() },
   522→      })
   523→      return { providerStatus: null, mapped: null, error: err.message }
   524→    }
   525→    throw err
   526→  }
   527→}

   529→async function completeSettlement(payment: PaymentRequest, providerStatus: string, chainStatus: string) {
   530→  const org = await db.organization.findUniqueOrThrow({ where: { id: payment.orgId } })
   531→  const network = (await db.wallet.findUniqueOrThrow({ where: { id: payment.sourceWalletId! } })).network
   532→  const sig = await db.transactionSignature.findFirst({ where: { paymentId: payment.id, confirmedAt: { not: null } } })
   533→  const providerRef = (await db.providerTransfer.findUnique({ where: { paymentId: payment.id } }))?.providerTransferId ?? null
   534→  await db.$transaction(async (tx) => {
   535→    await tx.settlementRecord.upsert({
   536→      where: { paymentId: payment.id },
   537→      create: { paymentId: payment.id, expectedBase: payment.amountBaseUnits, observedBase: payment.amountBaseUnits, providerStatus, chainStatus, signature: sig?.signature ?? null, reconciledAt: new Date(), mismatch: false },
   538→      update: { providerStatus, chainStatus, reconciledAt: new Date(), signature: sig?.signature ?? null, mismatch: false },
   539→    })
   540→    const existing = await tx.ledgerEntry.findFirst({ where: { paymentId: payment.id, entryType: 'PRINCIPAL' } })
   541→    if (!existing) {
   542→      await tx.ledgerEntry.create({
   543→        data: {
   544→          orgId: payment.orgId,
   545→          paymentId: payment.id,
   546→          entryType: 'PRINCIPAL',
   547→          direction: 'DEBIT',
   548→          mint: payment.mint,
   549→          amountBaseUnits: payment.amountBaseUnits,
   550→          signature: sig?.signature ?? null,
   551→          source: 'OBSERVED',
   552→          occurredAt: new Date(),
   553→        },
   554→      })
   555→    }
   556→  })
   557→  await db.reconciliationRecord.upsert({
   558→    where: { paymentId: payment.id },
   559→    create: { orgId: payment.orgId, paymentId: payment.id, providerRef, providerStatus, chainEvidenceJson: JSON.stringify({ chainStatus, signature: sig?.signature, network }), expectedBase: payment.amountBaseUnits, observedBase: payment.amountBaseUnits, state: 'MATCHED', detail: `Settled on ${network}` },
   560→    update: { providerStatus, state: 'MATCHED', detail: `Settled on ${network}`, lastCheckedAt: new Date() },
   561→  })
   562→  const fresh = await db.paymentRequest.findUniqueOrThrow({ where: { id: payment.id } })
   563→  await transition(fresh, 'COMPLETED', { reason: `Provider ${providerStatus}; chain ${chainStatus}` })
   564→  await db.paymentRequest.update({ where: { id: payment.id }, data: { completedAt: new Date() } })
   565→  if (payment.recipientId) {
   566→    await db.recipient.update({ where: { id: payment.recipientId }, data: { lastPaymentAt: new Date() } })
   567→  }
   568→  await audit({ orgId: payment.orgId, actorType: 'SYSTEM', action: 'PAYMENT_SETTLED', resourceType: 'PaymentRequest', resourceId: payment.id, data: { network, providerStatus, chainStatus } })
   569→  await notify(payment.orgId, { type: 'PAYMENT_COMPLETED', title: `Payment ${payment.reference} settled`, body: 'Verified against provider and chain evidence.', link: `#/app/payments/${payment.id}` })
   570→}
